class Solution {
    public List<Integer> leaders(int[] nums)
    {
        ArrayList<Integer>list=new ArrayList<>();
        if(nums.length==0)
        {
            return list;
        }

        int max=nums[nums.length-1];
        list.add(nums[nums.length-1]);

        for(int i=nums.length-2;i>=0;i--)
        {
            if(nums[i]>max)
            {
                max=nums[i];
                list.add(nums[i]);
            }
        }

        Collections.reverse(list);

        return list;
        
    }
}
